#mdmvegchange

mdm_change_veg_to_jungle = {
	is_foreign = no
	max_levels = 1
	pop_type = laborers
	
	employment_size = 0.001
	category = infrastructure_category
	
	megalopolis = yes
	metropolis = yes
	major_city = yes
	city = yes
	town = yes
	village = yes
	rural_settlement = yes

	price = mdm_default_price
	build_time = mdm_build_time_10_years
	
	forbidden_for_estates = yes
	
	location_potential = {
		climate = tropical #can only make jungle in area with enough rain
		OR = {
			NOR = { #cant convert to jungle if it is already jungle
				has_location_modifier = mdm_forest_to_jungle
				has_location_modifier = mdm_woods_to_jungle
				has_location_modifier = mdm_grasslands_to_jungle
				has_location_modifier = mdm_farmland_to_jungle
				has_location_modifier = mdm_sparse_to_jungle
				has_location_modifier = mdm_desert_to_jungle
			}
			AND = { #can also convert back to jungle if it was once jungle
				vegetation = jungle
				OR = {
					has_location_modifier = mdm_jungle_to_forest
					has_location_modifier = mdm_jungle_to_woods
					has_location_modifier = mdm_jungle_to_grasslands
					has_location_modifier = mdm_jungle_to_farmland
					has_location_modifier = mdm_jungle_to_sparse
					has_location_modifier = mdm_jungle_to_desert
				}
			}
		}
	}

	unique_production_methods = {
		mdm_change_veg_to_jungle_maintenance = {
			tools = 0.1
			category = building_maintenance
		}
	}

	on_built = {
		location = {
			mdm_clear_location_vegetation_modifier_effect = yes
			if = {
				limit = {
					vegetation = jungle
				}
				#do nothing, effect already cleared
			}
			else_if = {
				limit = {
					vegetation = forest
				}
				add_location_modifier = mdm_forest_to_jungle
			}
			else_if = {
				limit = {
					vegetation = woods
				}
				add_location_modifier = mdm_woods_to_jungle
			}
			else_if = {
				limit = {
					vegetation = grasslands
				}
				add_location_modifier = mdm_grasslands_to_jungle
			}
			else_if = {
				limit = {
					vegetation = farmland
				}
				add_location_modifier = mdm_farmland_to_jungle
			}
			else_if = {
				limit = {
					vegetation = sparse
				}
				add_location_modifier = mdm_sparse_to_jungle
			}
			else_if = {
				limit = {
					vegetation = desert
				}
				add_location_modifier = mdm_desert_to_jungle
			}
		}
	}

	construction_demand = mdm_change_veg_to_jungle_construction

	remove_if = {
		location = {	
			OR = {
					has_location_modifier = mdm_forest_to_jungle
					has_location_modifier = mdm_woods_to_jungle
					has_location_modifier = mdm_grasslands_to_jungle
					has_location_modifier = mdm_farmland_to_jungle
					has_location_modifier = mdm_sparse_to_jungle
					has_location_modifier = mdm_desert_to_jungle
				}
		}
	}
	
}
