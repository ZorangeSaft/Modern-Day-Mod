#mdmvegchange

mdm_change_veg_to_jungle = {
	is_foreign = no
	max_levels = 1
	pop_type = laborers
	
	employment_size = 0.001
	category = infrastructure_category
	
	megalopolis = yes
	metropolis = yes
	major_city = yes
	city = yes
	town = yes
	village = yes
	rural_settlement = yes

	price = mdm_default_price
	build_time = mdm_build_time_10_years
	
	forbidden_for_estates = yes
	
	location_potential = {
		climate = tropical #can only make jungle in area with enough rain
		mdm_is_jungle = no #cant build if it is already jungle
	}

	unique_production_methods = {
		mdm_change_veg_to_jungle_maintenance = {
			tools = 0.1
			category = building_maintenance
		}
	}

	on_built = {
		location = {
			mdm_clear_location_vegetation_modifier_effect = yes
			mdm_set_veg_to_jungle = yes
			add_location_modifier = {
				modifier = mdm_remove_temp_veg_building
				months = 1
			}
		}
	}

	construction_demand = mdm_change_veg_to_jungle_construction

	remove_if = {
		location = {	
			has_location_modifier = mdm_remove_temp_veg_building
		}
	}
	
}

mdm_change_veg_to_farmland = {
	is_foreign = no
	max_levels = 1
	pop_type = laborers
	
	employment_size = 0.001 
	category = infrastructure_category
	
	megalopolis = yes
	metropolis = yes
	major_city = yes
	city = yes
	town = yes
	village = yes
	rural_settlement = yes

	price = mdm_default_price
	build_time = mdm_build_time_10_years
	
	forbidden_for_estates = yes
	
	location_potential = {
		NOR = {
			climate = arctic #too cold
			topography = mountains #too mountainous
			topography = hills #too hilly
		}
		mdm_is_farmland = no #cant build if it is already farmland
		OR = {
			has_river = yes #needs water
			AND = {
				is_adjacent_to_lake = yes #needs water
				NOT = {
					any_neighbor_location = {
						topography = salt_pans #cant farm with salt water
					}
				}
				
			}
			topography = wetlands #were very often drained to make farmland
			AND = {
				climate = mediterranean #just very good climate for farming
				topography = flatland #so you cant just build it everywhere mediteranean
			}
		}
	}

	unique_production_methods = {
		mdm_change_veg_to_farmland_maintenance = {
			tools = 0.1
			category = building_maintenance
		}
	}

	on_built = {
		location = {
			mdm_clear_location_vegetation_modifier_effect = yes
			mdm_set_veg_to_farmland = yes
		}
	}

	construction_demand = mdm_change_veg_to_farmland_construction

	remove_if = {
		location = {	
			OR = {
					has_location_modifier = mdm_jungle_to_farmland
					has_location_modifier = mdm_forest_to_farmland
					has_location_modifier = mdm_woods_to_farmland
					has_location_modifier = mdm_grasslands_to_farmland
					has_location_modifier = mdm_sparse_to_farmland
					has_location_modifier = mdm_desert_to_farmland
				}
		}
	}
	
}
