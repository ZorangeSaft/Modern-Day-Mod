#mdmvegchange

mdm_change_veg_to_jungle = {
	is_foreign = no
	max_levels = 1
	pop_type = laborers
	
	employment_size = 0.001
	category = infrastructure_category
	
	megalopolis = yes
	metropolis = yes
	major_city = yes
	city = yes
	town = yes
	village = yes
	rural_settlement = yes

	price = mdm_default_price
	build_time = mdm_build_time_10_years
	
	forbidden_for_estates = yes
	
	location_potential = {
		OR = {
			AND = {
				climate = tropical #can only make jungle in area with enough rain
				mdm_has_jungle_vegetation = no #cant convert to jungle if it is already jungle
			}
			AND = { #can also convert back to jungle if it was once jungle
				vegetation = jungle
				mdm_has_original_vegetation = no
			}
		}
	}

	unique_production_methods = {
		mdm_change_veg_to_jungle_maintenance = {
			tools = 0.1
			category = building_maintenance
		}
	}

	on_built = {
		location = {
			mdm_change_vegetation_to = { TYPE = jungle }
		}
	}

	construction_demand = mdm_change_veg_to_jungle_construction

	remove_if = {
		location = {	
			mdm_has_jungle_vegetation = yes
			mdm_has_original_vegetation = no
		}
	}
	
}
