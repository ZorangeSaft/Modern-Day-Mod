#changed vegetation

wsf_is_jungle = {
	OR = { #it is ether vegetation based jungle or modifier based jungle, converted from something else
		vegetation = jungle
		has_location_modifier = wsf_forest_to_jungle
		has_location_modifier = wsf_woods_to_jungle
		has_location_modifier = wsf_grasslands_to_jungle
		has_location_modifier = wsf_farmland_to_jungle
		has_location_modifier = wsf_sparse_to_jungle
		has_location_modifier = wsf_desert_to_jungle
	}
	NOR = { #it is not jungle that has been converted away by a modifier. doesnt need to check for vegetation = jungle as well, 
			#because jungle_to_x should only ever apply to jungles, otherwise something went wrong
		has_location_modifier = wsf_jungle_to_forest
		has_location_modifier = wsf_jungle_to_woods
		has_location_modifier = wsf_jungle_to_grasslands
		has_location_modifier = wsf_jungle_to_farmland
		has_location_modifier = wsf_jungle_to_sparse
		has_location_modifier = wsf_jungle_to_desert
	}
}
