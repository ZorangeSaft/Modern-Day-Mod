#continents

#every continent has legumes, fruit and livestock
#only farmable and husbandry/grazing goods that give good in game have been added, and also horses for when making grasslands

wsf_is_in_europe_equivalent = { #has wheat, sturdy grains, legumes, fruit, olives, livestock, wool, horses
	OR = {
		continent = continent:europe
		sub_continent = sub_continent:north_africa #basegame north africa is included in europe instead of africa because of the similar rgos
	}
}

wsf_is_in_asia_equivalent = { #has wheat, sturdy grains, legumes, fruit, rice, olives, livestock, wool, horses
	continent = continent:asia
}

wsf_is_in_africa_equivalent = { #has wheat, sturdy grains, legumes, fruit, rice, livestock
	continent = continent:africa
	NOT = {
		sub_continent = sub_continent:north_africa #basegame north africa is not included with africa because of the similar rgos it has to europe
	}
}

wsf_is_in_north_america_equivalent = { #has maize, legumes, fruit, livestock in basegame
	sub_continent = sub_continent:north_america #technically a subcontinent in game
}

wsf_is_in_south_america_equivalent = { #has maize, legumes, fruit, potatoes, livestock in basegame
	sub_continent = sub_continent:south_america #technically a subcontinent in game, seperated because only the south has potatoes
}

wsf_is_in_australasia_equivalent = { #has sturdy grains, legumes, fruit, livestock
	continent = continent:australasia
}
