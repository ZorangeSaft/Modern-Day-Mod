#wsfvegchange


#set vegetation via static modifiers

wsf_set_veg_to_jungle = { #gives the correct modifier to change the vegetation via modifier based on what the defined normal vegetation is
	if = {
		limit = {
			vegetation = jungle
		}
		#do nothing, it is already jungle
	}
	else_if = {
		limit = {
			vegetation = forest
		}
		add_location_modifier = wsf_forest_to_jungle
	}
	else_if = {
		limit = {
			vegetation = woods
		}
		add_location_modifier = wsf_woods_to_jungle
		}
	else_if = {
		limit = {
			vegetation = grasslands
		}
		add_location_modifier = wsf_grasslands_to_jungle
	}
	else_if = {
		limit = {
			vegetation = farmland
		}
		add_location_modifier = wsf_farmland_to_jungle
	}
	else_if = {
		limit = {
			vegetation = sparse
		}
		add_location_modifier = wsf_sparse_to_jungle
	}
	else_if = {
		limit = {
			vegetation = desert
		}
		add_location_modifier = wsf_desert_to_jungle
	}
}
