#mdmvegchange

mdm_set_veg_to_jungle = {
	if = {
		limit = {
			vegetation = jungle
		}
		#do nothing, it is already jungle
	}
	else_if = {
		limit = {
			vegetation = forest
		}
		add_location_modifier = mdm_forest_to_jungle
	}
	else_if = {
		limit = {
			vegetation = woods
		}
		add_location_modifier = mdm_woods_to_jungle
		}
	else_if = {
		limit = {
			vegetation = grasslands
		}
		add_location_modifier = mdm_grasslands_to_jungle
	}
	else_if = {
		limit = {
			vegetation = farmland
		}
		add_location_modifier = mdm_farmland_to_jungle
	}
	else_if = {
		limit = {
			vegetation = sparse
		}
		add_location_modifier = mdm_sparse_to_jungle
	}
	else_if = {
		limit = {
			vegetation = desert
		}
		add_location_modifier = mdm_desert_to_jungle
	}
}
