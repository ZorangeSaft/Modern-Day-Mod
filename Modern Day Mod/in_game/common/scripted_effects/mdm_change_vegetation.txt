
#location scope
mdm_change_vegetation_to = {
	custom_description = {
		text = mdm_change_vegetation_to_text

		mdm_clear_location_vegetation_modifier_effect = yes
		set_local_variable = {
			name = new_vegetation
			value = flag:$TYPE$
		}
		if = {
			limit = { NOT = { vegetation = $TYPE$ } }
			switch = {
				trigger = vegetation
				jungle = {
					add_location_modifier = mdm_jungle_to_$TYPE$
				}
				forest = {
					add_location_modifier = mdm_forest_to_$TYPE$
				}
				woods = {
					add_location_modifier = mdm_woods_to_$TYPE$
				}
				grasslands = {
					add_location_modifier = mdm_grasslands_to_$TYPE$
				}
				farmland = {
					add_location_modifier = mdm_farmland_to_$TYPE$
				}
				sparse = {
					add_location_modifier = mdm_sparse_to_$TYPE$
				}
				desert = {
					add_location_modifier = mdm_desert_to_$TYPE$
				}
			}
		}
	}
}
