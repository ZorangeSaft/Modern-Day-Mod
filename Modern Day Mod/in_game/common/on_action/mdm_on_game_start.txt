on_game_start = {
  on_actions = {
    mdm_on_game_start
  }
}

mdm_on_game_start = {
  effect = {
    every_location_in_the_world = {
      limit = { is_land = yes }
      #geography based stuff
      set_variable = {
        name = mdm_local_wind_speed
        value = mdm_wind_turbine_max_building_level
      }
      set_variable = {
        name = mdm_local_solar_illumination
        value = mdm_solar_panel_max_building_level
      }
      set_variable = {
			  name = mdm_local_urbanisation_suitability
			  value = mdm_local_urbanisation_suitability_level
			}
		}

    #minerals
    scripted_geography:ore_deposit_aluminium_1000 = {
      every_location_in_scripted_geography = {
        set_variable = {
			    name = mdm_aluminium_ore_deposit_size
			    value = 1000
			  }
      }
    }
  }
}
