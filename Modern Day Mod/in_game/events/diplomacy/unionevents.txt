namespace = unionevents
# Random events
#Country breaks away from Union due to Ruler claiming power
unionevents.100 = {
	type = country_event
	title = unionevents.100.title
	desc = unionevents.100.desc
	option = {
		name = unionevents.100.a
		if = {
			limit = { exists = scope:root_country }
			show_as_tooltip = {
				union ?= {
					remove_country_from_international_organization = scope:root_country
				}
			}
			add_casus_belli = {
				target = scope:root_country
				type = casus_belli:cb_claim_throne
			}
		}
	}
}
