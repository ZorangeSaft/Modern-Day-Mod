namespace = mdm_debug_refresh_mapmodes

mdm_debug_refresh_mapmodes.1 = {
	type = country_event 				 # type of event 
	title = mdm_debug_refresh_mapmodes.title 			 # title key to be localized
	desc = mdm_debug_refresh_mapmodes.desc 			 # description key to be localized 
	historical_info = mdm_debug_refresh_mapmodes.historical_info  # optional description text to provide historical context
	
	trigger = { 	# List of requirements that need to be fulfilled for event to file
		#a
	}

	immediate = {
		error_log = "test message"
		every_location_in_the_world = {
			limit = { is_land = yes }
			set_variable = {
				name = mdm_local_wind_speed
				value = mdm_wind_turbine_max_building_level
			}
			set_variable = {
				name = mdm_local_solar_illumination
				value = mdm_solar_panel_max_building_level
			}
			set_variable = {
				name = mdm_local_urbanisation_suitability
				value = mdm_local_urbanisation_suitability_level
			}
		}
	}

	option = {
	name = mdm_debug_refresh_mapmodes.b
	}
}
